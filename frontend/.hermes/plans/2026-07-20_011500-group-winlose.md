# Group Tracking & Win/Lose Implementation Plan

> **For Hermes:** Use subagent-driven-development to implement this plan task-by-task.

**Goal:** Add visual solid/stripe group tracking for each player, pocketed-balls display, and game-over detection (win/lose) for both debug and multiplayer modes.

**Architecture:** The server already has a complete rules engine (`GameService.apply_rules`) that assigns groups and detects game-over. ```typescript
/**
 * Lightweight 8-ball rules for local/debug mode.
 * Mirrors the server's GameService.apply_rules for game-over detection only.
 */

export type Group = 'solids' | 'stripes' | null;

export interface ShotResult {
  gameOver: boolean;
  winner: number | null;   // player number (1 or 2)
  reason: string;
  groupsAssigned: boolean;
}

export function evaluateShot(
  pocketed: number[],
  cuePocketed: boolean,
  player1Group: Group,
  player2Group: Group,
  currentPlayer: number,
  firstContact: number | null,
  isBreak: boolean,
): ShotResult {
  const result: ShotResult = {
    gameOver: false,
    winner: null,
    reason: '',
    groupsAssigned: false,
  };

  const legalPockets = pocketed.filter(b => b !== 0 && b !== 8);
  const eightBallPocketed = pocketed.includes(8);
  const opponent = currentPlayer === 1 ? 2 : 1;

  let currentGroup: Group = currentPlayer === 1 ? player1Group : player2Group;

  // Determine group if not yet assigned (first pocket after break)
  if (currentGroup === null && legalPockets.length > 0) {
    const firstBall = legalPockets[0];
    currentGroup = (firstBall >= 1 && firstBall <= 7) ? 'solids' : 'stripes';
    result.groupsAssigned = true;
  }

  // Count remaining balls of shooter's group
  let remainingOwnGroup = 0;
  if (currentGroup) {
    this.balls.forEach((bd) => {
      const n = bd.number;
      if (n === 0 || n === 8) return;
      const g = (n >= 1 && n <= 7) ? 'solids' : 'stripes';
      if (g === currentGroup) remainingOwnGroup++;
    });
  }

  // 8-ball pocketed scenarios
  if (eightBallPocketed) {
    if (currentGroup === null) {
      // 8-ball on open table = loss
      result.gameOver = true;
      result.winner = opponent;
      result.reason = 'early_8_ball';
    } else if (remainingOwnGroup === 0) {
      // Cleared own group + pocketed 8-ball = win (unless also scratched)
      if (cuePocketed) {
        result.gameOver = true;
        result.winner = opponent;
        result.reason = 'scratch_on_8_ball';
      } else {
        result.gameOver = true;
        result.winner = currentPlayer;
        result.reason = 'pocketed_8_ball';
      }
    } else {
      // Early 8-ball = loss
      result.gameOver = true;
      result.winner = opponent;
      result.reason = 'early_8_ball';
    }
  }

  return result;
}

**Step 1 (revised): Create `src/rules.ts`**

```typescript
export type Group = 'solids' | 'stripes' | null;

export interface ShotEval {
  gameOver: boolean;
  winner: number | null;   // 1 or 2
  reason: string;
  player1Group: Group;
  player2Group: Group;
  groupsJustAssigned: boolean;
}

/**
 * Evaluate a shot for game-over conditions.
 * Does NOT count remaining balls — caller provides ownRemaining.
 */
export function evaluateShot(
  pocketed: number[],
  cuePocketed: boolean,
  currentPlayer: number,
  ownRemaining: number,          // balls of shooter's group still on table
  player1Group: Group,
  player2Group: Group,
): ShotEval {

  const result: ShotEval = {
    gameOver: false,
    winner: null,
    reason: '',
    player1Group,
    player2Group,
    groupsJustAssigned: false,
  };

  const legalPockets = pocketed.filter(b => b !== 0 && b !== 8);
  const eightBallPocketed = pocketed.includes(8);
  const opponent = currentPlayer === 1 ? 2 : 1;

  let currentGroup: Group = currentPlayer === 1 ? player1Group : player2Group;

  // Auto-assign groups if first pocket after break
  if (currentGroup === null && legalPockets.length > 0) {
    const firstBall = legalPockets[0];
    currentGroup = (firstBall >= 1 && firstBall <= 7) ? 'solids' : 'stripes';
    if (currentPlayer === 1) {
      result.player1Group = currentGroup;
      result.player2Group = currentGroup === 'solids' ? 'stripes' : 'solids';
    } else {
      result.player2Group = currentGroup;
      result.player1Group = currentGroup === 'solids' ? 'stripes' : 'solids';
    }
    result.groupsJustAssigned = true;
  }

  // 8-ball pocketed
  if (eightBallPocketed) {
    if (currentGroup === null) {
      result.gameOver = true;
      result.winner = opponent;
      result.reason = 'early_8_ball';
    } else if (ownRemaining <= 0) {
      if (cuePocketed) {
        result.gameOver = true;
        result.winner = opponent;
        result.reason = 'scratch_on_8_ball';
      } else {
        result.gameOver = true;
        result.winner = currentPlayer;
        result.reason = 'pocketed_8_ball';
      }
    } else {
      result.gameOver = true;
      result.winner = opponent;
      result.reason = 'early_8_ball';
    }
  }

  return result;
}
```

## Risks & Tradeoffs

- **Local rules vs server rules**: The local rules engine (`rules.ts`) is intentionally minimal — it only detects game-over. Full foul/group-assignment logic stays on the server for multiplayer. This means local and multiplayer modes might diverge slightly in edge cases (e.g., illegal break detection).
- **Pocketed-ball ownership**: In multiplayer, the server knows who shot, but the client needs to track pocketed balls per player. We use the `player` field from `shot_result` to attribute pocketed balls. In local mode, we use `myPlayerNum`.
- **Card height**: Increasing `cardH` from 50 to ~64 makes room for the extra info rows. This encroaches slightly on the play area but is acceptable.
